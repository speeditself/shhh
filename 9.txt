1. Create a simple index on cust_id and item_name
// ======================================================
db.orders.createIndex({cust_id:1})
db.orders.createIndex({item_name:1})
// Show Indexes
db.orders.getIndexes()
// ======================================================
// 2. Display all customer names with no repetition
// ======================================================
db.orders.distinct("cust_name")
// ======================================================
// 3. Find total no of orders and pending orders
// ======================================================
db.orders.aggregate([
{
    $group:{
        _id:null,
        total_orders:{$sum:1},
        pending_orders:{
            $sum:{
                $cond:[{$eq:["$status","Pending"]},1,0]
            }
        }
    }
}
])
// ======================================================
// 4. Sorting documents based on amount
// ======================================================
// WITHOUT INDEXING
db.orders.find().sort({amount:1}).explain("executionStats")
// Create Index on amount
db.orders.createIndex({amount:1})
// WITH INDEXING
db.orders.find().sort({amount:1}).explain("executionStats")

5. Pending orders details with and without indexing
// ======================================================
// WITHOUT INDEXING
db.orders.find({status:"Pending"}).explain("executionStats")
// Create Index on status
db.orders.createIndex({status:1})
// WITH INDEXING
db.orders.find({status:"Pending"}).explain("executionStats")
