// Create a simple index on cust_id and item_name

// Question 2
// Display all customer ids with no repetition

db.orders.distinct("cust_id")


// Question 3
// Find total no of orders and how many are pending

db.orders.aggregate([
{
    $group:
    {
        _id:null,
        total_orders:{$sum:1},
        pending_orders:
        {
            $sum:
            {
                $cond:[{$eq:["$status","P"]},1,0]
            }
        }
    }
}
])

// Question 6
// Show how many orders are placed by each customer

db.orders.aggregate([
{
    $group:
    {
        _id:"$cust_id",
        total_orders:{$sum:1}
    }
}
])


// Question 7
// Display customer ids and total pending amount in descending order

db.orders.aggregate([

{
    $match:{status:"P"}
},

{
    $group:
    {
        _id:"$cust_id",
        total_pending_amount:{$sum:"$amount"}
    }
},

{
    $sort:{total_pending_amount:-1}
}

])


// Question 8
// Display customer ids in ascending order with total delivered amount

db.orders.aggregate([

{
    $match:{status:"D"}
},

{
    $group:
    {
        _id:"$cust_id",
        total_delivered_amount:{$sum:"$amount"}
    }
},

{
    $sort:{_id:1}
}

])


// Question 9
// Show top three selling items

db.orders.aggregate([

{
    $group:
    {
        _id:"$item_name",
        total_quantity:{$sum:"$quantity"}
    }
},

{
    $sort:{total_quantity:-1}
},

{
    $limit:3
}

])


// Question 10
// Find the date on which maximum orders are received

db.orders.aggregate([

{
    $group:
    {
        _id:"$DtOfOrder",
        total_orders:{$sum:1}
    }
},

{
    $sort:{total_orders:-1}
},

{
    $limit:1
}

])


// Question 11
// List customer ids with FIRST and LAST order date

db.orders.aggregate([

{
    $group:
    {
        _id:"$cust_id",
        first_order:{$min:"$DtOfOrder"},
        last_order:{$max:"$DtOfOrder"}
    }
},

{
    $sort:{_id:1}
}

])
